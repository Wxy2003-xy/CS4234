cmake_minimum_required(VERSION 3.20)
project(networkflow VERSION 1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(NETWORKFLOW_BUILD_APP "Build the SDL2/OpenGL visualizer" ON)

add_library(flow_core src/flow.cpp)
target_include_directories(flow_core PUBLIC include)
if(MSVC)
  target_compile_options(flow_core PRIVATE /W4)
else()
  target_compile_options(flow_core PRIVATE -Wall -Wextra -Wpedantic)
endif()

if(NETWORKFLOW_BUILD_APP)
  find_package(SDL2 CONFIG REQUIRED)
  find_package(OpenGL REQUIRED)
  add_executable(networkflow src/main.cpp)
  target_link_libraries(networkflow PRIVATE flow_core SDL2::SDL2 OpenGL::GL)
  target_compile_definitions(networkflow PRIVATE SDL_MAIN_HANDLED GL_SILENCE_DEPRECATION)
  if(NOT MSVC)
    target_compile_options(networkflow PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(flow_tests tests/flow_tests.cpp)
  target_link_libraries(flow_tests PRIVATE flow_core)
  add_test(NAME flow_tests COMMAND flow_tests)
endif()
